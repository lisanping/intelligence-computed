theta = init_random()
for epoch in range(epochs):
    for batch in data:
        pred = batch['x'] @ theta
        loss = MSE(pred, batch['y'])
        grad = compute_grad(loss, theta)  # 链式法则
        theta -= lr * grad
    plot_loss(loss)
